Supabase設定 & SQL

接続設定について

SupabaseのプロジェクトURLとAnon Keyを設定するとクラウドと同期します。未設定の場合はブラウザ内に保存（デモモード）されます。

-- 1. テーブル作成
create table schedule_items (
  id uuid default gen_random_uuid() primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  date date not null,
  subject text not null,
  type text not null check (type in ('小テスト', '授業変更')),
  content text not null,
  author text not null
);

-- 2. セキュリティポリシー設定 (匿名アクセス許可)
alter table schedule_items enable row level security;
create policy "Allow anonymous select" on schedule_items for select using (true);
create policy "Allow anonymous insert" on schedule_items for insert with check (true);
create policy "Allow anonymous delete" on schedule_items for delete using (true);
